import csv
import io
from flask import Blueprint, jsonify, request, Response
from ..constants import Role
from ..services.order_service import (
    product_wise_report,
    date_wise_report,
    get_dashboard_summary,
)
from ..utils.decorators import role_required

reports_bp = Blueprint("reports", __name__, url_prefix="/api/reports")


@reports_bp.get("/summary")
@role_required(Role.ADMIN.value)
def dashboard_summary():
    return jsonify(get_dashboard_summary()), 200


@reports_bp.get("/product-wise")
@role_required(Role.ADMIN.value)
def product_wise():
    data = product_wise_report()
    if request.args.get("format") == "csv":
        return _generate_product_csv(data)
    return jsonify(data), 200

@reports_bp.get("/date-wise")
@role_required(Role.ADMIN.value)
def date_wise():
    group = request.args.get("group", "day")
    if group not in ("day", "month"):
        return jsonify(error="group must be 'day' or 'month'"), 400
    data = date_wise_report(group)
    if request.args.get("format") == "csv":
        return _generate_date_csv(data, group)
    return jsonify(data), 200

def _generate_product_csv(data):
    output = io.StringIO()
    writer = csv.writer(output)
    writer.writerow(["Product", "Items Sold", "Units Sold", "Revenue (INR)"])
    for row in data:
        writer.writerow([row["product"], row["items"], row["units"], f"{row['revenue']:.2f}"])
    return Response(
        output.getvalue(),
        mimetype="text/csv",
        headers={"Content-Disposition": "attachment; filename=product_wise_report.csv"},
    )

def _generate_date_csv(data, group):
    output = io.StringIO()
    writer = csv.writer(output)
    writer.writerow([f"Period ({group})", "Orders", "Units Sold", "Revenue (INR)"])
    for row in data:
        writer.writerow([row["period"][:10], row["orders"], row["units"], f"{row['revenue']:.2f}"])
    return Response(
        output.getvalue(),
        mimetype="text/csv",
        headers={"Content-Disposition": f"attachment; filename=date_wise_report_{group}.csv"},
    )

import logging
from flask import Blueprint, g, jsonify, request
from sqlalchemy.exc import IntegrityError
from ..constants import Role
from ..models import Order
from ..services.order_service import (
    VALID_STATUSES,
    change_status,
    create_order as create_order_service,
    find_duplicates,
    visible_orders_for,
)
from ..utils.decorators import auth_required, role_required
from ..utils.permissions import can_view_order
log = logging.getLogger(__name__)

orders_bp = Blueprint("orders", __name__, url_prefix="/api/orders")

def _json():
    return request.get_json(silent=True) or {}

def _order_response(order, include_items=False):
    data = order.to_dict(include_items=include_items)
    if include_items:
        data["total"] = float(
            sum(i.quantity * i.unit_price for i in order.items) if order.items else 0
        )
    return data

@orders_bp.get("/duplicates")
@role_required(Role.ADMIN.value)
def list_duplicates():
    return jsonify(duplicates=find_duplicates()), 200

@orders_bp.get("")
@auth_required
def list_orders():
    user = g.current_user
    q = visible_orders_for(user)

    status = request.args.get("status")
    if status:
        if status not in VALID_STATUSES:
            return jsonify(error=f"Invalid status: {status}"),400
        q = q.filter(Order.order_status == status)

    customer_id = request.args.get("customer_id")
    if customer_id:
        if user.role != Role.ADMIN.value and (
            user.customer is None or user.customer.customer_id != customer_id
        ):
            return jsonify(error="Forbidden"), 403
        q = q.filter(Order.customer_id == customer_id)

    q = q.order_by(Order.order_date.desc(), Order.order_id.desc())

    page = max(request.args.get("page", 1, type=int), 1)
    per_page = min(max(request.args.get("per_page", 50, type=int), 1), 200)

    paginated = q.paginate(page=page, per_page=per_page, error_out=False)

    return jsonify(
        orders=[
            {
                **_order_response(o),
                "total": float(sum(i.quantity * i.unit_price for i in o.items)),
                "item_count": len(o.items)
            } for o in paginated.items
        ],
        page=paginated.page,
        per_page=paginated.per_page,
        total=paginated.total,
        pages=paginated.pages,
    ), 200

@orders_bp.get("/<order_id>")
@auth_required
def get_order(order_id):
    user = g.current_user
    order = Order.query.filter_by(order_id=order_id).first()
    if order is None:
        return jsonify(error="Order not found"), 404
    if not can_view_order(user, order):
        return jsonify(error="Order not found"), 404
    return jsonify(_order_response(order, include_items=True)), 200

@orders_bp.post("")
@role_required(Role.ADMIN.value)
def create_order():
    payload = _json()
    try:
        order, err, status = create_order_service(payload)
    except IntegrityError:
        return jsonify(error="Order already exists"), 409
    except Exception:
        log.exception("create_order failed")
        return jsonify(error="Failed to create order"), 400

    if err:
        if isinstance(err, list):
            return jsonify(errors=err), status
        return jsonify(error=err), status

    return jsonify(_order_response(order, include_items=True)), status

@orders_bp.patch("/<order_id>/status")
@role_required(Role.ADMIN.value)
def update_status(order_id):
    new_status = _json().get("order_status")
    if not new_status:
        return jsonify(error="order_status is required"), 400

    order = Order.query.filter_by(order_id=order_id).first()
    if order is None:
        return jsonify(error="Order not found"), 404

    ok, msg = change_status(order, new_status)
    if not ok:
        return jsonify(error=msg), 400

    return jsonify(_order_response(order)), 200

@products_bp.get("")
@auth_required
def list_products():
    products = Product.query.order_by(Product.pro_name).all()
    return jsonify(products=[p.to_dict() for p in products]), 200

@products_bp.post("")
@role_required(Role.ADMIN.value)
def create_product():
    data = request.get_json() or {}
    name = (data.get("pro_name") or "").strip()
    price = data.get("current_price")

    if not name:
        return jsonify(error="pro_name is required"), 400
    if price is None or float(price) < 0:
        return jsonify(error="current_price must be >= 0"), 400
    if Product.query.filter_by(pro_name=name).first():
        return jsonify(error="Product name already exists"), 409

    product = Product(pro_name=name, current_price=price)
    db.session.add(product)
    db.session.commit()
    return jsonify(product=product.to_dict()), 201

@products_bp.patch("/<int:product_id>")
@role_required(Role.ADMIN.value)
def update_product_price(product_id):
    product = db.session.get(Product, product_id)
    if not product:
        return jsonify(error="Product not found"), 404
        
    new_price = request.get_json().get("current_price")
    if new_price is None or float(new_price) < 0:
        return jsonify(error="Invalid price"), 400
        
    product.current_price = new_price
    db.session.commit()
    return jsonify(message="Price updated", product=product.to_dict()), 200

@products_bp.delete("/<int:product_id>")
@role_required(Role.ADMIN.value)
def delete_product(product_id):
    product = db.session.get(Product, product_id)
    if not product:
        return jsonify(error="Product not found"), 404
    
    if product.order_items:
        return jsonify(error="Cannot delete product: it is used in orders. Archive it instead."), 409
    
    db.session.delete(product)
    db.session.commit()
    return jsonify(message="Product deleted"), 200

import re
from datetime import date
from decimal import Decimal
from sqlalchemy import func
from ..extensions import db
from ..models import Customer, Order, OrderItem, Product
from ..constants import Role

VALID_STATUSES = {"Pending", "Received", "Shipped", "Delivered", "Cancelled", "Returned"}
ALLOWED_TRANSITIONS = {
    "Pending":   {"Received", "Shipped", "Cancelled"},
    "Received":  {"Shipped", "Cancelled"},
    "Shipped":   {"Delivered", "Returned"},
    "Delivered": {"Returned"},
    "Cancelled": set(),
    "Returned":  set(),
}
SALE_STATUSES = frozenset({"Shipped", "Delivered"})

def validate_order_payload(payload):
    errors = []

    order_id = payload.get("order_id")
    if not order_id or not re.match(r"^ORD\d{4}$", str(order_id)):
        errors.append("order_id must match ORD####")

    order_date = payload.get("order_date")
    if not order_date:
        errors.append("order_date is required")
    else:
        try:
            date.fromisoformat(str(order_date))
        except (TypeError, ValueError):
            errors.append("order_date must be YYYY-MM-DD")

    customer_id = payload.get("customer_id")
    if not customer_id:
        errors.append("customer_id is required")
    elif not re.match(r"^C\d{4}$", str(customer_id)):
        errors.append("customer_id must match C####")
    elif not Customer.query.filter_by(customer_id=str(customer_id)).first():
        errors.append(f"Unknown customer: {customer_id}")

    items = payload.get("items") or []
    if not items:
        errors.append("At least one item is required")

    seen = set()
    for i, item in enumerate(items):
        pid = item.get("product_id")
        if pid in seen:
            errors.append(f"items[{i}].product_id is duplicated")
        seen.add(pid)
       
        if pid is None:
            errors.append(f"items[{i}].product_id is required")
        else:
            prod = db.session.get(Product, pid)
            if prod is None:
                errors.append(f"items[{i}].product_id does not exist")
            elif prod.current_price is None:
                errors.append(f"items[{i}].product_id has no price set")

        qty = item.get("quantity")
        if qty is None:
            errors.append(f"items[{i}].quantity is required")
        else:
            try:
                if not isinstance(qty, int) or isinstance(qty, bool):
                    raise ValueError
                if qty < 1 or qty > 1000:
                    errors.append(f"items[{i}].quantity must be 1..1000")
            except ValueError:
                errors.append(f"items[{i}].quantity must be an integer")

    status = payload.get("order_status", "Pending")
    if status not in VALID_STATUSES:
        errors.append(f"Invalid order_status: {status!r}")

    return errors

def create_order(payload):
    errors = validate_order_payload(payload)
    if errors:
        return None, errors, 400

    order_id = payload["order_id"]

    if db.session.get(Order, order_id):
        return None, f"Order {order_id} already exists", 409

    order = Order(
        order_id=order_id,
        customer_id=payload["customer_id"],
        order_date=date.fromisoformat(payload["order_date"]),
        order_status="Pending",
    )
    db.session.add(order)

    try:
        for item in payload.get("items", []):
            product = db.session.get(Product, item["product_id"])
            db.session.add(OrderItem(
                order_id=order.order_id,
                product_id=item["product_id"],
                quantity=item["quantity"],
                unit_price=product.current_price, 
            ))
        db.session.commit()
    except Exception:
        db.session.rollback()
        raise

    return order, None, 201

def visible_orders_for(user):
    if user is None:
        return Order.query.filter(False)
    if user.role == Role.ADMIN.value:
        return Order.query
    return (
        Order.query
        .join(Customer, Customer.customer_id == Order.customer_id)
        .filter(Customer.user_id == user.id)
    )

def can_transition(old_status, new_status):
    if old_status not in VALID_STATUSES or new_status not in VALID_STATUSES:
        return False
    return new_status in ALLOWED_TRANSITIONS.get(old_status, set())

def change_status(order, new_status):
    if not can_transition(order.order_status, new_status):
        return False, f"Cannot move from {order.order_status} to {new_status}"
    order.order_status = new_status
    db.session.commit()
    return True, None

def product_wise_report():
    rows = (
        db.session.query(
            Product.pro_name.label("product"),
            func.count(OrderItem.order_item_id).label("items"),
            func.sum(OrderItem.quantity).label("units"),
            func.sum(OrderItem.quantity * OrderItem.unit_price).label("revenue"),
        )
        .join(OrderItem, OrderItem.product_id == Product.product_id)
        .join(Order, Order.order_id == OrderItem.order_id)
        .filter(Order.order_status.in_(SALE_STATUSES))
        .group_by(Product.pro_name)
        .order_by(func.sum(OrderItem.quantity * OrderItem.unit_price).desc())
        .all()
    )
    return [
        {
            "product": r.product,
            "items": int(r.items or 0),
            "units": float(r.units or 0),
            "revenue": float(r.revenue or 0),
        }
        for r in rows
    ]

def date_wise_report(group="day"):
    bucket = func.date_trunc("month", Order.order_date) if group == "month" else Order.order_date

    rows = (
        db.session.query(
            bucket.label("bucket"),
            func.count(func.distinct(Order.order_id)).label("orders"),
            func.sum(OrderItem.quantity).label("units"),
            func.sum(OrderItem.quantity * OrderItem.unit_price).label("revenue"),
        )
        .join(OrderItem, OrderItem.order_id == Order.order_id)
        .filter(Order.order_status.in_(SALE_STATUSES))
        .group_by(bucket)
        .order_by(bucket)
        .all()
    )
    return [
        {
            "period": r.bucket.isoformat() if hasattr(r.bucket, "isoformat") else str(r.bucket),
            "orders": int(r.orders or 0),
            "units": float(r.units or 0),
            "revenue": float(r.revenue or 0),
        }
        for r in rows
    ]

def find_duplicates():
    rows = (
        db.session.query(
            Order.customer_id,
            OrderItem.product_id,
            OrderItem.quantity,
            OrderItem.unit_price,
            Order.order_date,
            func.count(OrderItem.order_item_id).label("occurrences"),
            func.array_agg(Order.order_id).label("order_ids"),
        )
        .join(OrderItem, OrderItem.order_id == Order.order_id)
        .group_by(
            Order.customer_id,
            OrderItem.product_id,
            OrderItem.quantity,
            OrderItem.unit_price,
            Order.order_date,
        )
        .having(func.count(OrderItem.order_item_id) > 1)
        .order_by(func.count(OrderItem.order_item_id).desc())
        .all()
    )
    return [
        {
            "customer_id": r.customer_id,
            "product_id": r.product_id,
            "quantity": int(r.quantity) if r.quantity is not None else None,
            "unit_price": float(r.unit_price) if r.unit_price is not None else None,
            "order_date": r.order_date.isoformat() if r.order_date is not None else None,
            "occurrences": int(r.occurrences or 0),
            "order_ids": sorted(r.order_ids or []),
        }
        for r in rows
    ]

def get_dashboard_summary():
    
    revenue = db.session.query(func.sum(OrderItem.quantity * OrderItem.unit_price))\
        .join(Order, Order.order_id == OrderItem.order_id)\
        .filter(Order.order_status.in_(SALE_STATUSES)).scalar() or 0.0

    sales_orders = db.session.query(func.count(func.distinct(Order.order_id)))\
        .filter(Order.order_status.in_(SALE_STATUSES)).scalar() or 0

    units_sold = db.session.query(func.sum(OrderItem.quantity))\
        .join(Order, Order.order_id == OrderItem.order_id)\
        .filter(Order.order_status.in_(SALE_STATUSES)).scalar() or 0

    awaiting_action = db.session.query(func.count(Order.order_id))\
        .filter(Order.order_status.in_(["Pending", "Received"])).scalar() or 0

    latest_orders = Order.query.order_by(Order.created_at.desc()).limit(5).all()
    recent = []
    for o in latest_orders:
        total = sum(i.quantity * i.unit_price for i in o.items)
        recent.append({
            "order_id": o.order_id,
            "customer_id": o.customer_id,
            "order_date": o.order_date.isoformat() if o.order_date else None,
            "order_status": o.order_status,
            "total": float(total),
        })

    duplicates = find_duplicates()[:3]

    product_rows = (
        db.session.query(
            Product.pro_name.label("product"),
            func.count(OrderItem.order_item_id).label("items"),
            func.sum(OrderItem.quantity).label("units"),
            func.sum(OrderItem.quantity * OrderItem.unit_price).label("revenue"),
        )
        .join(OrderItem, OrderItem.product_id == Product.product_id)
        .join(Order, Order.order_id == OrderItem.order_id)
        .filter(Order.order_status.in_(SALE_STATUSES))
        .group_by(Product.pro_name)
        .order_by(func.sum(OrderItem.quantity * OrderItem.unit_price).desc())
        .all()
    )
    product_breakdown = [
        {
            "product": r.product,
            "items": int(r.items or 0),
            "units": float(r.units or 0),
            "revenue": float(r.revenue or 0),
        }
        for r in product_rows
    ]

    monthly_rows = (
        db.session.query(
            func.date_trunc("month", Order.order_date).label("bucket"),
            func.count(func.distinct(Order.order_id)).label("orders"),
            func.sum(OrderItem.quantity).label("units"),
            func.sum(OrderItem.quantity * OrderItem.unit_price).label("revenue"),
        )
        .join(OrderItem, OrderItem.order_id == Order.order_id)
        .filter(Order.order_status.in_(SALE_STATUSES))
        .group_by("bucket")
        .order_by("bucket")
        .all()
    )
    monthly_breakdown = [
        {
            "period": r.bucket.isoformat() if hasattr(r.bucket, "isoformat") else str(r.bucket),
            "orders": int(r.orders or 0),
            "units": float(r.units or 0),
            "revenue": float(r.revenue or 0),
        }
        for r in monthly_rows
    ]

    return {
        "revenue": float(revenue),
        "sales_orders": int(sales_orders),
        "units_sold": int(units_sold),
        "awaiting_action": int(awaiting_action),
        "latest_orders": recent,
        "duplicate_groups": duplicates,
        "product_breakdown": product_breakdown,
        "monthly_breakdown": monthly_breakdown,
    }

import logging
import os
import sys
from logging.handlers import RotatingFileHandler
from dotenv import load_dotenv

BASE_DIR = os.path.abspath(os.path.join(os.path.dirname(__file__), ".."))

_env_name = os.getenv("FLASK_ENV", "development")
_env_file = ".env.test" if _env_name == "test" else ".env"
load_dotenv(os.path.join(BASE_DIR, _env_file))

def configure_logging(app):
    log_dir = os.path.join(BASE_DIR, "logs")
    os.makedirs(log_dir, exist_ok=True)

    formatter = logging.Formatter(
        "%(asctime)s | %(levelname)-7s | %(name)s | %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S",
    )

    console = logging.StreamHandler(sys.stdout)
    console.setFormatter(formatter)

    file_handler = RotatingFileHandler(
        os.path.join(log_dir, "app.log"),
        maxBytes=5 * 1024 * 1024,
        backupCount=3,
        encoding="utf-8",
    )
    file_handler.setFormatter(formatter)

    root = logging.getLogger()
    root.handlers = [console, file_handler]
    root.setLevel(logging.INFO)

    logging.getLogger("sqlalchemy.engine").setLevel(logging.WARNING)
    logging.getLogger("sqlalchemy.pool").setLevel(logging.WARNING)
    logging.getLogger("flask_jwt_extended").setLevel(logging.WARNING)
    logging.getLogger("werkzeug").setLevel(logging.INFO)

    if os.environ.get("WERKZEUG_RUN_MAIN") == "true":
        app.logger.info("Server ready")

    donutOptions.value.labels = topProducts.map(p => p.product)
    donutSeries.value = topProducts.map(p => p.revenue)
  } catch (err) {
    errorMsg.value = 'Failed to load dashboard data.'
  } finally {
    loading.value = false
  }
})

const downloadReport = (type, group = 'month') => {
  let url = `/api/reports/${type}?format=csv`
  if (type === 'date-wise') url += `&group=${group}`

  fetch(url, { headers: { Authorization: `Bearer ${auth.token}` } })
    .then(res => {
      if (!res.ok) throw new Error('Download failed')
      return res.blob()
    })
    .then(blob => {
      const link = document.createElement('a')
      link.href = URL.createObjectURL(blob)
      link.download = `${type}_report_${new Date().toISOString().slice(0, 10)}.csv`
      link.click()
      URL.revokeObjectURL(link.href)
    })
    .catch(err => {
      console.error(err)
      alert('Failed to download report')
    })
}
</script>

<script setup>
import { ref, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import api from '@/services/api'

const router = useRouter()
const products = ref([])
const customers = ref([])
const errorMsg = ref('')
const form = ref({
  order_id: '',
  customer_id: '',
  order_date: new Date().toISOString().split('T')[0],
  product_id: '',
  quantity: 1
})

onMounted(async () => {
  try {
    const [prodRes, custRes] = await Promise.all([
      api.get('/products'),
      api.get('/customers')
    ])
    
    products.value = prodRes.data.products
    customers.value = custRes.data.customers
    
    if (products.value.length) form.value.product_id = products.value[0].product_id
    if (customers.value.length) form.value.customer_id = customers.value[0].customer_id
  } catch (err) {
    errorMsg.value = 'Failed to load form data.'
  }
})

const handleSubmit = async () => {
  try {
    errorMsg.value = ''
    const payload = {
      order_id: form.value.order_id,
      customer_id: form.value.customer_id,
      order_date: form.value.order_date,
      items: [{ product_id: form.value.product_id, quantity: form.value.quantity }]
    }
    await api.post('/orders', payload)
    router.push('/app/orders')
  } catch (err) {
    errorMsg.value = err.response?.data?.errors?.join(', ') || err.response?.data?.error || 'Failed to create order'
  }
}
</script>

import { createRouter, createWebHistory } from 'vue-router'
import { useAuthStore } from '@/stores/auth'

const routes = [
  { path: '/', name: 'Landing', component: () => import('@/views/Landing.vue') },
  { path: '/login', name: 'Login', component: () => import('@/views/Login.vue') },
  { path: '/register', name: 'Register', component: () => import('@/views/Register.vue') },
  { path: '/forgot-password', name: 'ForgotPassword', component: () => import('@/views/ForgotPassword.vue') },
  { path: '/reset-password', name: 'ResetPassword', component: () => import('@/views/ResetPassword.vue') },
  { 
    path: '/app', 
    component: () => import('@/layouts/AppLayout.vue'),
    meta: { requiresAuth: true },
    children: [
      { path: '', redirect: '/app/orders' },
      { path: 'orders', name: 'Orders', component: () => import('@/views/Orders.vue') },
      { path: 'orders/:id', name: 'OrderDetail', component: () => import('@/views/OrderDetail.vue') },
      { path: 'products', name: 'Products', component: () => import('@/views/Products.vue'), meta: { requiresAdmin: true } },
      { path: 'customers', name: 'Customers', component: () => import('@/views/Customers.vue'), meta: { requiresAdmin: true } },
      { path: 'reports', name: 'Reports', component: () => import('@/views/Reports.vue'), meta: { requiresAdmin: true } },
      { path: 'duplicates', name: 'Duplicates', component: () => import('@/views/Duplicates.vue'), meta: { requiresAdmin: true } },
      { path: 'new-order', name: 'CreateOrder', component: () => import('@/views/CreateOrder.vue'), meta: { requiresAdmin: true } },
    ]
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes,
})

router.beforeEach((to, from, next) => {
  const auth = useAuthStore()
  if (to.meta.requiresAuth && !auth.isAuthenticated) {
    next('/login')
  } 
  else if (to.meta.requiresAdmin && !auth.isAdmin) {
    next('/app/orders')
  } 
  else {
    next()
  }
})

export default router

import axios from 'axios'

const api = axios.create({ baseURL: '/api' })

api.interceptors.request.use((config) => {
  const token = localStorage.getItem('token')
  if (token) config.headers.Authorization = `Bearer ${token}`
  return config
})

api.interceptors.response.use(
  (res) => res,
  (err) => {
    if (err.response && (err.response.status === 401 || err.response.status === 422)) {
      localStorage.removeItem('token')
      localStorage.removeItem('user')
      if (window.location.pathname !== '/login' && window.location.pathname !== '/') {
        window.location.href = '/login'
      }
    }
    return Promise.reject(err)
  }
)

export default api

function getRecordKey(record) {
  if (record.replay_cycle_id && record.message_id) {
    return `${record.replay_cycle_id}:${record.message_id}`;
  }

  return String(record.id);
}

function compareRecords(a, b) {
  const timestampDifference =
    new Date(a.event_ts).getTime() -
    new Date(b.event_ts).getTime();

  if (timestampDifference !== 0) {
    return timestampDifference;
  }

  return Number(a.id) - Number(b.id);
}

function compareEventsNewestFirst(a, b) {
  const timestampDifference =
    new Date(b.event_ts).getTime() -
    new Date(a.event_ts).getTime();

  if (timestampDifference !== 0) {
    return timestampDifference;
  }

  return Number(b.id) - Number(a.id);
}

function getLatestById(records) {
  return records.reduce((latest, candidate) => {
    if (!latest) return candidate;

    return Number(candidate.id) >= Number(latest.id)
      ? candidate
      : latest;
  }, null);
}

    mergeRecords(records, { live = false } = {}) {
      if (!Array.isArray(records) || records.length === 0) {

      }

      const validRecords = records.filter(
        (record) =>
          record &&
          record.event_ts &&
          record.message_id &&
          record.replay_cycle_id &&
          record.payload &&
          record.id !== undefined &&
          record.id !== null
      );

      if (validRecords.length === 0) {
        return;
      }

      // Use the newest inserted record to identify this batch's cycle.
      const newestIncomingRecord = getLatestById(validRecords);
      const incomingCycleId = newestIncomingRecord.replay_cycle_id;

      // A cycle transition starts a new in-memory chart history.
      if (this.currentCycleId !== incomingCycleId) {
        this.history = [];
        this.latestReading = null;
        this.currentCycleId = incomingCycleId;
      }

      const cycleRecords = validRecords.filter(
        (record) => record.replay_cycle_id === this.currentCycleId

      const latestCandidates = [

          record.id !== undefined &&
          record.id !== null
      );

      this.latestReading = getLatestById(latestCandidates);

      if (live && this.latestReading) {

      }

const API_BASE_URL = (
  import.meta.env.VITE_API_BASE_URL || "http://127.0.0.1:8000"
).replace(/\/+$/, "");

async function request(path) {
  let response;

  try {
    response = await fetch(`${API_BASE_URL}${path}`);
  } catch {
    throw new Error("Unable to reach the Flask API server.");
  }

  let result;

  try {
    result = await response.json();
  } catch {
    throw new Error(`Invalid API response (${response.status}).`);
  }

  if (!response.ok) {
    throw new Error(
      result.error || `API request failed (${response.status}).`
    );
  }

  return result;
}

async function downloadCsv(path, fallbackFilename) {
  let response;

  try {
    response = await fetch(`${API_BASE_URL}${path}`);
  } catch {
    throw new Error("Unable to reach the Flask API server.");
  }

  if (!response.ok) {
    let message = `CSV export failed (${response.status}).`;

    try {
      const result = await response.json();
      message = result.error || message;
    } catch {
      // Keep the HTTP error message when the response is not JSON.
    }

    throw new Error(message);
  }

  const blob = await response.blob();
  const disposition = response.headers.get("Content-Disposition") || "";
  const filenameMatch = disposition.match(/filename="?([^";]+)"?/i);
  const filename = filenameMatch?.[1] || fallbackFilename;

  const objectUrl = URL.createObjectURL(blob);
  const link = document.createElement("a");

  link.href = objectUrl;
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  link.remove();

  URL.revokeObjectURL(objectUrl);
}

function addHistoricalScope(params, { cycleId, start, end, hourStart } = {}) {
  if (cycleId) params.set("cycle_id", cycleId);
  if (start) params.set("start", start);
  if (end) params.set("end", end);
  if (hourStart) params.set("hour_start", hourStart);
  return params;
}

export const telemetryApi = {
  getHealth() {
    return request("/api/health");
  },

  getLatest(lineId = "line30") {
    const params = new URLSearchParams({ line_id: lineId });
    return request(`/api/telemetry/latest?${params}`);
  },

  getHistory({ lineId = "line30", limit = 100, cycleId, start, end } = {}) {
    const params = new URLSearchParams({ line_id: lineId, limit: String(limit) });
    addHistoricalScope(params, { cycleId, start, end });
    return request(`/api/telemetry/history?${params}`);
  },

  getStatistics({ lineId = "line30", field, start, end } = {}) {
    if (!field) throw new Error("A telemetry field is required for statistics.");
    const params = new URLSearchParams({ line_id: lineId, field });
    if (start) params.set("start", start);
    if (end) params.set("end", end);
    return request(`/api/telemetry/statistics?${params}`);
  },

  getHourlySummary({ lineId = "line30", hours, cycleId, hourStart } = {}) {
    const params = new URLSearchParams({ line_id: lineId });
    if (hours !== undefined && hours !== null) params.set("hours", String(hours));
    addHistoricalScope(params, { cycleId, hourStart });
    return request(`/api/summary/hourly?${params}`);
  },

  getEvents({ lineId = "line30", limit = 100, cycleId, start, end } = {}) {
    const params = new URLSearchParams({ line_id: lineId, limit: String(limit) });
    addHistoricalScope(params, { cycleId, start, end });
    return request(`/api/events?${params}`);
  },

  getReplayCycles({ lineId = "line30", limit = 50 } = {}) {
    const params = new URLSearchParams({ line_id: lineId, limit: String(limit) });
    return request(`/api/history/cycles?${params}`);
  },

  downloadHourlySummary({ lineId = "line30", hours, cycleId, hourStart } = {}) {
    const params = new URLSearchParams({ line_id: lineId });
    if (hours !== undefined && hours !== null) params.set("hours", String(hours));
    addHistoricalScope(params, { cycleId, hourStart });
    return downloadCsv(`/api/exports/hourly-summary.csv?${params}`, "line30-hourly-summary.csv");
  },

  downloadReadings({ lineId = "line30", cycleId, start, end } = {}) {
    const params = new URLSearchParams({ line_id: lineId });
    addHistoricalScope(params, { cycleId, start, end });
    return downloadCsv(`/api/exports/readings.csv?${params}`, "line30-historical-readings.csv");
  },

  downloadEvents({ lineId = "line30", cycleId, start, end } = {}) {
    const params = new URLSearchParams({ line_id: lineId });
    addHistoricalScope(params, { cycleId, start, end });
    return downloadCsv(`/api/exports/events.csv?${params}`, "line30-event-history.csv");
  },
};

function formatValue(value, digits = 2) {
  if (value === null || value === undefined || value === "") return "â€”";
  const number = Number(value);
  return Number.isFinite(number) ? number.toFixed(digits) : "â€”";

function formatDuration(value) {
  const seconds = Number(value);
  if (!Number.isFinite(seconds) || seconds <= 0) return "0s";
  const total = Math.floor(seconds);
  const hours = Math.floor(total / 3600);
  const minutes = Math.floor((total % 3600) / 60);
  const remaining = total % 60;
  if (hours) return `${hours}h ${String(minutes).padStart(2, "0")}m ${String(remaining).padStart(2, "0")}s`;
  if (minutes) return `${minutes}m ${String(remaining).padStart(2, "0")}s`;
  return `${remaining}s`;
}
function formatTimestamp(value) {
  if (!value) return "â€”";

import json
import logging

import paho.mqtt.client as mqtt
from sqlalchemy.exc import SQLAlchemyError

from server.repositories.telemetry_repository import save_telemetry
from server.sockets.broadcaster import (
    broadcast_operational_events,
    broadcast_telemetry,
)

logger = logging.getLogger(__name__)


def _ack_message(client, message):
    result = client.ack(message.mid, message.qos)

    if result != mqtt.MQTT_ERR_SUCCESS:
        logger.error(
            "MQTT acknowledgement failed: packet_id=%s result=%s",
            message.mid,
            result,
        )
        return False

    return True


def on_message(client, userdata, message):
    try:
        raw_payload = message.payload.decode("utf-8")
        payload = json.loads(raw_payload)
    except (UnicodeDecodeError, json.JSONDecodeError) as exc:
        logger.error(
            "Discarding malformed MQTT message: topic=%s error=%s",
            message.topic,
            exc,
        )
        _ack_message(client, message)
        return

    try:
        record = save_telemetry(payload)
    except ValueError as exc:
        logger.error(
            "Discarding invalid telemetry payload: topic=%s error=%s",
            message.topic,
            exc,
        )
        _ack_message(client, message)
        return
    except SQLAlchemyError:
        logger.exception(
            "Database persistence failed. MQTT message remains "
            "unacknowledged: topic=%s packet_id=%s",
            message.topic,
            message.mid,
        )
        return
    except Exception:
        logger.exception(
            "Unexpected telemetry processing failure. MQTT message "
            "remains unacknowledged: topic=%s packet_id=%s",
            message.topic,
            message.mid,
        )
        return

    if record is None:
        logger.info(
            "Duplicate telemetry ignored: message_id=%s",
            payload.get("message_id"),
        )
        _ack_message(client, message)
        return

    logger.info(
        "Telemetry persisted: record_id=%s message_id=%s "
        "generated_events=%s",
        record["id"],
        payload.get("message_id"),
        len(record["events"]),
    )

    acknowledged = _ack_message(client, message)

    if not acknowledged:
        logger.error(
            "Telemetry was persisted but MQTT acknowledgement "
            "failed: record_id=%s",
            record["id"],
        )

    try:
        broadcast_telemetry(record, payload)
    except Exception:
        logger.exception(
            "Telemetry was persisted but Socket.IO broadcast failed: "
            "record_id=%s",
            record["id"],
        )

    try:
        broadcast_operational_events(record)
    except Exception:
        logger.exception(
            "Telemetry was persisted but operational event "
            "broadcast failed: record_id=%s",
            record["id"],
        )

import logging

import paho.mqtt.client as mqtt

from server.config import (
    MQTT_CLIENT_ID,
    MQTT_HOST,
    MQTT_KEEPALIVE,
    MQTT_PORT,
    MQTT_QOS,
    MQTT_TOPIC,
)
from server.mqtt.message_handler import on_message

logger = logging.getLogger(__name__)
_mqtt_client = None


def is_mqtt_connected() -> bool:
    return (
        _mqtt_client is not None
        and _mqtt_client.is_connected()
    )


def on_connect(client, userdata, flags, reason_code, properties):
    if reason_code.is_failure:
        logger.error(
            "MQTT connection rejected: %s",
            reason_code,
        )
        return

    logger.info(
        "Connected to MQTT broker: %s:%s",
        MQTT_HOST,
        MQTT_PORT,
    )

    result, message_id = client.subscribe(
        MQTT_TOPIC,
        qos=MQTT_QOS,
    )

    if result != mqtt.MQTT_ERR_SUCCESS:
        logger.error(
            "MQTT subscription failed: return_code=%s",
            result,
        )
        return

    logger.info(
        "Subscribed to topic '%s' at QoS %s",
        MQTT_TOPIC,
        MQTT_QOS,
    )


def create_mqtt_client():
    client = mqtt.Client(
        callback_api_version=mqtt.CallbackAPIVersion.VERSION2,
        client_id=MQTT_CLIENT_ID,
        protocol=mqtt.MQTTv311,
        clean_session=False,
        manual_ack=True,
    )

    client.on_connect = on_connect
    client.on_message = on_message

    client.reconnect_delay_set(
        min_delay=1,
        max_delay=30,
    )

    return client


def start_mqtt_subscriber():
    global _mqtt_client

    client = create_mqtt_client()

    logger.info(
        "Starting MQTT subscriber: client_id=%s",
        MQTT_CLIENT_ID,
    )

    client.connect(
        MQTT_HOST,
        MQTT_PORT,
        MQTT_KEEPALIVE,
    )

    _mqtt_client = client
    client.loop_start()

    return client


def stop_mqtt_subscriber(client):
    global _mqtt_client

    if client is None:
        return

    logger.info("Stopping MQTT subscriber")

    client.disconnect()
    client.loop_stop()

    if _mqtt_client is client:
        _mqtt_client = None

EVENT_HISTORY_SQL = text("""
    SELECT
        id,
        line_id,
        event_type,
        event_code,
        severity,
        state,
        title,
        description,
        event_ts,
        recorded_at,
        reading_id,
        replay_cycle_id,
        details
    FROM telemetry.events
    WHERE line_id = :line_id
        AND replay_cycle_id = COALESCE(
          :selected_cycle_id,
          (
              SELECT replay_cycle_id
              FROM telemetry.readings
              WHERE line_id = :line_id
              ORDER BY received_at DESC, id DESC
              LIMIT 1
          )
        )
      AND (:start_time IS NULL OR event_ts >= :start_time)
      AND (:end_time IS NULL OR event_ts < :end_time)
    ORDER BY event_ts DESC, id DESC
    LIMIT :row_limit
    """).bindparams(
        bindparam("selected_cycle_id", type_=UUID(as_uuid=False)),
        bindparam("start_time", type_=DateTime()),
        bindparam("end_time", type_=DateTime()),
    )

    limit: int = 100,

    start=None,
    end=None,

    with engine.connect() as connection:

            {
                "line_id": line_id,
                "selected_cycle_id": cycle_id,
                "start_time": start,
                "end_time": end,
                "row_limit": limit,
            },
        ).mappings().all()

========== server\repositories\replay_repository.py ==========
from sqlalchemy import text
from server.database import engine

REPLAY_CYCLES_SQL = text("""
    WITH cycle_statistics AS (
        SELECT
            replay_cycle_id,
            COUNT(*) AS persisted_records,

            COUNT(*) FILTER (
                WHERE payload ? 'replay_total_records'
            ) AS metadata_records,

            COUNT(*) FILTER (
                WHERE payload ->> 'source_file_sha256'
                    ~ '^[0-9a-fA-F]{64}$'
            ) AS hash_records,
            COUNT(*) FILTER (
                WHERE payload ? 'replay_session_id'
                  AND NULLIF(BTRIM(payload ->> 'replay_session_id'), '') IS NOT NULL
            ) AS session_id_records,

            COUNT(DISTINCT payload ->> 'replay_session_id')
                AS session_id_variants,

            COUNT(*) FILTER (
                WHERE payload ->> 'replay_cycle_number' ~ '^[0-9]+$'
            ) AS cycle_number_records,

            COUNT(DISTINCT payload ->> 'replay_cycle_number')
            FILTER (
                WHERE payload ->> 'replay_cycle_number' ~ '^[0-9]+$'
            ) AS cycle_number_variants,

            COUNT(*) FILTER (
                WHERE payload ? 'source_row'
                  AND payload ->> 'source_row' ~ '^[0-9]+$'
            ) AS valid_source_rows,

            COUNT(*) FILTER (
                WHERE payload ->> 'replay_total_records' ~ '^[0-9]+$'
            ) AS expected_count_records,

            COUNT(DISTINCT (payload ->> 'replay_total_records')::BIGINT)
            FILTER (
                WHERE payload ->> 'replay_total_records' ~ '^[0-9]+$'
            ) AS expected_count_variants,

            MAX((payload ->> 'replay_total_records')::BIGINT)
            FILTER (
                WHERE payload ->> 'replay_total_records' ~ '^[0-9]+$'
            ) AS expected_records,

            COUNT(DISTINCT payload ->> 'source_file_sha256')
            FILTER (
                WHERE payload ->> 'source_file_sha256'
                    ~ '^[0-9a-fA-F]{64}$'
            ) AS source_hash_variants,

            MAX(payload ->> 'source_file_sha256')
            FILTER (
                WHERE payload ->> 'source_file_sha256'
                    ~ '^[0-9a-fA-F]{64}$'
            ) AS source_file_sha256,

            COUNT(DISTINCT (payload ->> 'source_row')::BIGINT)
            FILTER (
                WHERE payload ->> 'source_row' ~ '^[0-9]+$'
            ) AS distinct_source_rows,

            MIN((payload ->> 'source_row')::BIGINT)
            FILTER (
                WHERE payload ->> 'source_row' ~ '^[0-9]+$'
            ) AS min_source_row,

            MAX((payload ->> 'source_row')::BIGINT)
            FILTER (
                WHERE payload ->> 'source_row' ~ '^[0-9]+$'
            ) AS max_source_row,

            MAX(payload ->> 'replay_session_id') AS replay_session_id,
            MAX(payload ->> 'replay_cycle_number') AS replay_cycle_number,

            MIN(event_ts) AS first_event_ts,
            MAX(event_ts) AS last_event_ts,
            MIN(received_at) AS first_received_at,
            MAX(received_at) AS last_received_at

        FROM telemetry.readings
        WHERE line_id = :line_id
        GROUP BY replay_cycle_id
    ),

    classified_cycles AS (
        SELECT
            *,
            CASE
                WHEN metadata_records = 0
                    THEN 'UNKNOWN'

                WHEN metadata_records <> persisted_records
                  OR hash_records <> persisted_records
                  OR session_id_records <> persisted_records
                  OR session_id_variants <> 1
                  OR cycle_number_records <> persisted_records
                  OR cycle_number_variants <> 1
                  OR valid_source_rows <> persisted_records
                  OR expected_count_variants <> 1
                  OR source_hash_variants <> 1
                    THEN 'INCOMPLETE'

                WHEN expected_records = persisted_records
                  AND distinct_source_rows = expected_records
                  AND min_source_row = 2
                  AND max_source_row = expected_records + 1
                    THEN 'COMPLETED'

                ELSE 'INCOMPLETE'
            END AS status
        FROM cycle_statistics
    )

    SELECT
        replay_cycle_id,
        replay_session_id,
        replay_cycle_number,
        status,

        persisted_records,
        expected_records,
        distinct_source_rows,
        min_source_row,
        max_source_row,
        source_file_sha256,

        first_event_ts,
        last_event_ts,
        first_received_at,
        last_received_at,

        CASE
            WHEN status = 'COMPLETED'
                THEN 'All expected source records are persisted.'
            WHEN status = 'UNKNOWN'
                THEN 'Legacy cycle has no replay completion metadata.'
            WHEN metadata_records <> persisted_records
                THEN 'Some persisted records are missing replay metadata.'
            WHEN hash_records <> persisted_records
                THEN 'Some persisted records are missing the source fingerprint.'
            WHEN session_id_records <> persisted_records
                THEN 'Some persisted records are missing the replay session ID.'
            WHEN session_id_variants <> 1
                THEN 'Replay session IDs are inconsistent within this cycle.'
            WHEN cycle_number_records <> persisted_records
                THEN 'Some persisted records are missing a valid replay cycle number.'
            WHEN cycle_number_variants <> 1
                THEN 'Replay cycle numbers are inconsistent within this cycle.'
            WHEN valid_source_rows <> persisted_records
                THEN 'Some persisted records have invalid or missing source rows.'
            WHEN expected_count_variants <> 1
                THEN 'Expected record counts are missing or inconsistent.'
            WHEN source_hash_variants <> 1
                THEN 'Source file fingerprints are missing or inconsistent.'
            WHEN expected_records <> persisted_records
                THEN 'Persisted record count does not match the expected count.'
            WHEN distinct_source_rows <> expected_records
                THEN 'Distinct source-row coverage does not match the expected count.'
            WHEN min_source_row <> 2
              OR max_source_row <> expected_records + 1
                THEN 'Source-row boundaries do not match the expected range.'
            ELSE 'Replay completion checks have not been satisfied.'
        END AS status_reason

    FROM classified_cycles
    ORDER BY last_received_at DESC, replay_cycle_id DESC
    LIMIT :limit
""")


def get_replay_cycles(line_id: str, limit: int = 50) -> list[dict]:
    with engine.connect() as connection:
        rows = connection.execute(
            REPLAY_CYCLES_SQL,
            {
                "line_id": line_id,
                "limit": limit,
            },
        ).mappings().all()
    return [dict(row) for row in rows]

from datetime import datetime, timedelta
from sqlalchemy import DateTime, bindparam, text
from sqlalchemy.dialects.postgresql import UUID
from server.database import engine

CYCLE_INFO_SQL = text("""
    SELECT
        replay_cycle_id,
        MIN(event_ts) AS first_event_ts,
        MAX(event_ts) AS last_event_ts
    FROM telemetry.readings
    WHERE line_id = :line_id
      AND replay_cycle_id = COALESCE(
          :selected_cycle_id,
          (
              SELECT replay_cycle_id
              FROM telemetry.readings
              WHERE line_id = :line_id
              ORDER BY received_at DESC, id DESC
              LIMIT 1
          )
      )
    GROUP BY replay_cycle_id
    """).bindparams(
        bindparam("selected_cycle_id", type_=UUID(as_uuid=False)),
    )

HOURLY_SUMMARY_SQL = text("""
WITH
        buckets AS (
            SELECT generate_series(
                :window_start,
                :window_end,
                INTERVAL '1 hour'
            ) AS hour_start
        ),

        cycle_readings AS (
            SELECT
                id,
                event_ts,
                CASE
                    WHEN jsonb_typeof(
                        payload -> 'Burner Output Load'
                    ) = 'number'
                    THEN (payload ->> 'Burner Output Load')::NUMERIC
                END AS burner_load,

                CASE
                    WHEN jsonb_typeof(payload -> 'Burner PV') = 'number'
                    THEN (payload ->> 'Burner PV')::NUMERIC
                END AS burner_pv,

                CASE
                    WHEN jsonb_typeof(payload -> 'DeltaT') = 'number'
                    THEN (payload ->> 'DeltaT')::NUMERIC
                END AS delta_t,

                CASE
                    WHEN jsonb_typeof(payload -> 'Moisture PV') = 'number'
                    THEN (payload ->> 'Moisture PV')::NUMERIC
                END AS moisture,

                CASE
                    WHEN jsonb_typeof(payload -> 'Paddle Speed') = 'number'
                    THEN (payload ->> 'Paddle Speed')::NUMERIC
                END AS paddle_speed,

                CASE
                    WHEN jsonb_typeof(payload -> 'oil Level PV') = 'number'
                    THEN (payload ->> 'oil Level PV')::NUMERIC
                END AS oil_level

            FROM telemetry.readings
            WHERE line_id = :line_id
             AND replay_cycle_id = :cycle_id
        ),

        ordered_readings AS (
            SELECT
                id,
                event_ts,
                burner_load,
                burner_pv,
                delta_t,
                moisture,
                paddle_speed,
                oil_level,
                LEAD(event_ts) OVER (
                    ORDER BY event_ts, id
                ) AS next_event_ts
            FROM cycle_readings
        ),

        hourly_metrics AS (
            SELECT
                date_trunc('hour', event_ts) AS hour_start,
                COUNT(*) AS sample_count,

                AVG(burner_pv) AS burner_pv_avg,
                MIN(burner_pv) AS burner_pv_min,
                MAX(burner_pv) AS burner_pv_max,

                AVG(delta_t) AS delta_t_avg,
                MIN(delta_t) AS delta_t_min,
                MAX(delta_t) AS delta_t_max,

                AVG(burner_load) AS burner_load_avg,
                MIN(burner_load) AS burner_load_min,
                MAX(burner_load) AS burner_load_max,

                AVG(moisture) AS moisture_avg,
                MIN(moisture) AS moisture_min,
                MAX(moisture) AS moisture_max,

                AVG(paddle_speed) AS paddle_speed_avg,
                MIN(paddle_speed) AS paddle_speed_min,
                MAX(paddle_speed) AS paddle_speed_max,

                AVG(oil_level) AS oil_level_avg,
                MIN(oil_level) AS oil_level_min,
                MAX(oil_level) AS oil_level_max

            FROM ordered_readings
            WHERE event_ts >= :window_start
              AND event_ts < :window_end + INTERVAL '1 hour'
            GROUP BY date_trunc('hour', event_ts)
        ),

        burner_intervals AS (
            SELECT
                event_ts,
                next_event_ts
            FROM ordered_readings
            WHERE burner_load = 100
              AND next_event_ts > event_ts
              AND event_ts < :window_end + INTERVAL '1 hour'
              AND next_event_ts > :window_start
        ),

        burner_hour_splits AS (
            SELECT
                hour_bucket.hour_start,
                EXTRACT(
                    EPOCH FROM (
                        LEAST(
                            burner_intervals.next_event_ts,
                            hour_bucket.hour_start + INTERVAL '1 hour'
                        )
                        -
                        GREATEST(
                            burner_intervals.event_ts,
                            hour_bucket.hour_start
                        )
                    )
                ) AS duration_seconds
            FROM burner_intervals
            CROSS JOIN LATERAL generate_series(
                date_trunc(
                    'hour',
                    GREATEST(
                        burner_intervals.event_ts,
                        :window_start
                    )
                ),
                date_trunc(
                    'hour',
                    LEAST(
                        burner_intervals.next_event_ts,
                        :window_end + INTERVAL '1 hour'
                    ) - INTERVAL '1 microsecond'
                ),
                INTERVAL '1 hour'
            ) AS hour_bucket(hour_start)
        ),
        burner_durations AS (
            SELECT
                hour_start,
                SUM(duration_seconds) AS burner_at_100_seconds
            FROM burner_hour_splits
            GROUP BY hour_start
        )

        SELECT
            buckets.hour_start,
            COALESCE(metrics.sample_count, 0) AS sample_count,

            metrics.burner_pv_avg,
            metrics.burner_pv_min,
            metrics.burner_pv_max,

            metrics.delta_t_avg,
            metrics.delta_t_min,
            metrics.delta_t_max,

            metrics.burner_load_avg,
            metrics.burner_load_min,
            metrics.burner_load_max,

            metrics.moisture_avg,
            metrics.moisture_min,
            metrics.moisture_max,

            metrics.paddle_speed_avg,
            metrics.paddle_speed_min,
            metrics.paddle_speed_max,

            metrics.oil_level_avg,
            metrics.oil_level_min,
            metrics.oil_level_max,

            COALESCE(
                durations.burner_at_100_seconds,
                0
            ) AS burner_at_100_seconds

        FROM buckets
        LEFT JOIN hourly_metrics AS metrics
            ON metrics.hour_start = buckets.hour_start
        LEFT JOIN burner_durations AS durations
            ON durations.hour_start = buckets.hour_start
        ORDER BY buckets.hour_start ASC
        """).bindparams(
            bindparam("cycle_id", type_=UUID(as_uuid=False)),
            bindparam("window_start", type_=DateTime()),
            bindparam("window_end", type_=DateTime()),
        )

def get_hourly_summary_data(
    line_id: str,
    hours: int | None = None,
    cycle_id: str | None = None,
    hour_start: datetime | None = None,
) -> dict | None:
    """Fetch a complete cycle summary or one selected hourly bucket."""

    with engine.connect() as connection:
        cycle_info = connection.execute(
            CYCLE_INFO_SQL,
            {
                "line_id": line_id,
                "selected_cycle_id": cycle_id,
            },
        ).mappings().one_or_none()

        if cycle_info is None:
            return None

        cycle_id = cycle_info["replay_cycle_id"]
        first_event_ts = cycle_info["first_event_ts"]
        last_event_ts = cycle_info["last_event_ts"]

        if first_event_ts is None or last_event_ts is None:
            return None

        if hour_start is not None:
            window_start = hour_start.replace(
                minute=0,
                second=0,
                microsecond=0,
            )
            window_end = window_start

        elif hours is None:
            window_start = first_event_ts.replace(
                minute=0,
                second=0,
                microsecond=0,
            )
            window_end = last_event_ts.replace(
                minute=0,
                second=0,
                microsecond=0,
            )

        else:
            window_end = last_event_ts.replace(
                minute=0,
                second=0,
                microsecond=0,
            )
            window_start = window_end - timedelta(hours=hours - 1)

        parameters = {
            "line_id": line_id,
            "cycle_id": cycle_id,
            "window_start": window_start,
            "window_end": window_end,
        }

        hourly_rows = connection.execute(
            HOURLY_SUMMARY_SQL,
            parameters,
        ).mappings().all()

        statistics = connection.execute(
            SUMMARY_STATISTICS_SQL,
            parameters,
        ).mappings().one()

    return {
        "cycle_id": cycle_id,
        "window_start": window_start,
        "window_end": window_end,
        "hourly_rows": hourly_rows,
        "statistics": statistics,
    }

from sqlalchemy import DateTime, String, bindparam, text
from sqlalchemy.dialects.postgresql import UUID

LATEST_READING_SQL = text("""
    SELECT
        id,
        message_id,
        replay_session_id,
        replay_cycle_id,
        replay_cycle_number,
        line_id,
        event_ts,
        received_at,
        payload
    FROM telemetry.readings
    WHERE line_id = :line_id
    AND replay_cycle_id = (
        SELECT replay_cycle_id
        FROM telemetry.readings
        WHERE line_id = :line_id
        ORDER BY received_at DESC, id DESC
        LIMIT 1
    )
    ORDER BY event_ts DESC, id DESC

    """)

HISTORY_SQL = text("""
    SELECT
        id,
        message_id,
        replay_session_id,
        replay_cycle_id,
        replay_cycle_number,
        line_id,
        event_ts,
        received_at,
        payload
    FROM telemetry.readings
    WHERE line_id = :line_id
      AND replay_cycle_id = COALESCE(
          :selected_cycle_id,
          (
              SELECT replay_cycle_id
              FROM telemetry.readings
              WHERE line_id = :line_id
              ORDER BY received_at DESC, id DESC
              LIMIT 1
          )
      )
      AND (:start_time IS NULL OR event_ts >= :start_time)
      AND (:end_time IS NULL OR event_ts < :end_time)
      AND (:before_time IS NULL OR event_ts < :before_time)
    ORDER BY event_ts DESC, id DESC
    LIMIT :row_limit
    """).bindparams(
        bindparam("selected_cycle_id", type_=UUID(as_uuid=False)),
        bindparam("start_time", type_=DateTime()),
        bindparam("end_time", type_=DateTime()),
        bindparam("before_time", type_=DateTime()),
    )

STATISTICS_SQL = text("""
    WITH selected_cycle AS (
        SELECT COALESCE(
            :selected_cycle_id,
            (
                SELECT replay_cycle_id
                FROM telemetry.readings
                WHERE line_id = :line_id
                ORDER BY received_at DESC, id DESC
                LIMIT 1
            )
        ) AS replay_cycle_id
    )
    SELECT
        selected_cycle.replay_cycle_id AS cycle_id,

        COUNT(*) FILTER (
            WHERE jsonb_typeof(readings.payload -> :field_name) = 'number'
        ) AS sample_count,

        AVG((readings.payload ->> :field_name)::NUMERIC) FILTER (
            WHERE jsonb_typeof(readings.payload -> :field_name) = 'number'
        ) AS average,

        MIN((readings.payload ->> :field_name)::NUMERIC) FILTER (
            WHERE jsonb_typeof(readings.payload -> :field_name) = 'number'
        ) AS minimum,

        MAX((readings.payload ->> :field_name)::NUMERIC) FILTER (
            WHERE jsonb_typeof(readings.payload -> :field_name) = 'number'
        ) AS maximum

    FROM selected_cycle
    LEFT JOIN telemetry.readings AS readings
      ON readings.line_id = :line_id
     AND readings.replay_cycle_id = selected_cycle.replay_cycle_id
     AND (:start_time IS NULL OR readings.event_ts >= :start_time)
     AND (:end_time IS NULL OR readings.event_ts < :end_time)
    GROUP BY selected_cycle.replay_cycle_id
    """).bindparams(
        bindparam("selected_cycle_id", type_=UUID(as_uuid=False)),
        bindparam("field_name", type_=String()),
        bindparam("start_time", type_=DateTime()),
        bindparam("end_time", type_=DateTime()),
    )

def get_latest_reading(line_id: str):
    with engine.connect() as connection:
        return connection.execute(
            LATEST_READING_SQL,
            {"line_id": line_id},
        ).mappings().one_or_none()

def get_reading_history(
    line_id: str,
    before,
    limit: int,
    cycle_id=None,
    start=None,
    end=None,

def _validate_payload(payload: dict) -> dict:
    if not isinstance(payload, dict):
        raise ValueError("Telemetry payload must be a JSON object.")

    required_fields = (
        "message_id",
        "replay_session_id",
        "replay_cycle_id",
        "replay_cycle_number",
        "line_id",
        "event_ts",
    )

    missing = [
        field
        for field in required_fields
        if field not in payload
    ]

    if missing:
        raise ValueError(
            "Telemetry payload is missing fields: "
            + ", ".join(missing)
        )

    message_id = str(payload["message_id"]).strip()
    line_id = str(payload["line_id"]).strip()

    if not message_id:
        raise ValueError("message_id cannot be empty.")

    if not line_id:
        raise ValueError("line_id cannot be empty.")

    try:
        replay_session_id = PythonUUID(
            str(payload["replay_session_id"])
        )
        replay_cycle_id = PythonUUID(
            str(payload["replay_cycle_id"])
        )
    except (ValueError, TypeError, AttributeError) as exc:
        raise ValueError(
            "Invalid replay session or cycle UUID."
        ) from exc

    raw_cycle_number = payload["replay_cycle_number"]

    if isinstance(raw_cycle_number, bool):
        raise ValueError(
            "replay_cycle_number must be a positive integer."
        )

    try:
        cycle_number = int(raw_cycle_number)
    except (ValueError, TypeError, OverflowError) as exc:
        raise ValueError(
            "replay_cycle_number must be a positive integer."
        ) from exc

    if isinstance(raw_cycle_number, float):
        if not raw_cycle_number.is_integer():
            raise ValueError(
                "replay_cycle_number must be a positive integer."
            )

    if cycle_number <= 0:
        raise ValueError(
            "replay_cycle_number must be greater than zero."
        )
    if "replay_total_records" in payload:
        raw_total = payload["replay_total_records"]
        raw_source_row = payload.get("source_row")
        source_hash = payload.get("source_file_sha256")

        if (
            isinstance(raw_total, bool)
            or not isinstance(raw_total, int)
            or raw_total <= 0
        ):
            raise ValueError(
                "replay_total_records must be a positive integer."
            )

        if (
            isinstance(raw_source_row, bool)
            or not isinstance(raw_source_row, int)
            or not 2 <= raw_source_row <= raw_total + 1
        ):
            raise ValueError(
                "source_row must be an integer within the expected "
                "source dataset row range."
            )

        if (
            not isinstance(source_hash, str)
            or re.fullmatch(r"[0-9a-fA-F]{64}", source_hash) is None
        ):
            raise ValueError(
                "source_file_sha256 must be a valid 64-character "
                "SHA-256 hexadecimal digest."
            )
    try:
        event_ts = datetime.fromisoformat(
            str(payload["event_ts"])
        )
    except (ValueError, TypeError) as exc:
        raise ValueError(
            "event_ts must be a valid ISO 8601 timestamp."
        ) from exc

    if event_ts.tzinfo is not None:
        raise ValueError(
            "event_ts must be timezone-naive for the current "
            "database schema."
        )

    try:
        serialized_payload = json.dumps(
            payload,
            allow_nan=False,
            separators=(",", ":"),
        )
    except (ValueError, TypeError) as exc:
        raise ValueError(
            "Telemetry payload must contain valid JSON values."
        ) from exc

    return {
        "message_id": message_id,
        "replay_session_id": str(replay_session_id),
        "replay_cycle_id": str(replay_cycle_id),
        "replay_cycle_number": cycle_number,
        "line_id": line_id,
        "event_ts": event_ts,
        "payload": serialized_payload,
    }


def _get_moisture_value(payload: dict) -> float | None:
    value = payload.get("Moisture PV")

    if isinstance(value, bool):
        return None

    if not isinstance(value, (int, float)):
        return None

    value = float(value)

    if not math.isfinite(value):
        return None

    return value


def _get_moisture_event_code(value: float) -> str | None:
    if value < MOISTURE_LOW_LIMIT:
        return "MOISTURE_LOW"

    if value > MOISTURE_HIGH_LIMIT:
        return "MOISTURE_HIGH"

    return None


def _event_title(event_code: str) -> str:
    if event_code == "MOISTURE_LOW":
        return "Moisture Below Operating Limit"

    if event_code == "MOISTURE_HIGH":
        return "Moisture Above Operating Limit"

    return "Moisture Operating Range"


def _insert_alarm_event(
    connection,
    *,
    line_id: str,
    event_code: str,
    event_state: str,
    event_ts: datetime,
    reading_id: int,
    replay_cycle_id: str,
    moisture_value: float,
    previous_event_code: str | None,
    next_event_code: str | None,
) -> dict:

    if event_state == "ACTIVE":
        description = (
            f"Moisture PV is {moisture_value:.4f}. "
            f"Operating limits are "
            f"{MOISTURE_LOW_LIMIT:.2f}â€“"
            f"{MOISTURE_HIGH_LIMIT:.2f}."
        )
    else:
        description = (
            f"Moisture alarm {event_code} cleared. "
            f"Current Moisture PV is {moisture_value:.4f}."
        )

    details = {
        "moisture_value": moisture_value,
        "lower_limit": MOISTURE_LOW_LIMIT,
        "upper_limit": MOISTURE_HIGH_LIMIT,
        "previous_event_code": previous_event_code,
        "next_event_code": next_event_code,
    }

    result = connection.execute(
        INSERT_EVENT_SQL,
        {
            "line_id": line_id,
            "event_code": event_code,
            "state": event_state,
            "title": _event_title(event_code),
            "description": description,
            "event_ts": event_ts,
            "reading_id": reading_id,
            "replay_cycle_id": replay_cycle_id,
            "details": json.dumps(
                details,
                allow_nan=False,
                separators=(",", ":"),
            ),
        },
    )

    inserted = result.mappings().one()

    return {
        "id": inserted["id"],
        "line_id": line_id,
        "event_type": "ALARM",
        "event_code": event_code,
        "severity": "WARNING",
        "state": event_state,
        "title": _event_title(event_code),
        "description": description,
        "event_ts": event_ts.isoformat(),
        "recorded_at": inserted["recorded_at"].isoformat(),
        "reading_id": reading_id,
        "replay_cycle_id": replay_cycle_id,
        "details": details,
    }


def _process_moisture_alarm(
    connection,
    *,
    line_id: str,
    event_ts: datetime,
    reading_id: int,
    replay_cycle_id: str,
    moisture_value: float,
) -> list[dict]:

    new_event_code = _get_moisture_event_code(
        moisture_value
    )

    connection.execute(
        INSERT_ALARM_STATE_SQL,
        {
            "line_id": line_id,
            "rule_key": RULE_KEY,
            "replay_cycle_id": replay_cycle_id,
        },
    )

    current_state = connection.execute(
        LOCK_ALARM_STATE_SQL,
        {
            "line_id": line_id,
            "rule_key": RULE_KEY,
            "replay_cycle_id": replay_cycle_id,
        },
    ).mappings().one()

    current_event_code = (
        current_state["active_event_code"]
        if current_state["state"] == "ACTIVE"
        else None
    )

    events = []

    transitioned = current_event_code != new_event_code

    if transitioned and current_event_code is not None:
        events.append(
            _insert_alarm_event(
                connection,
                line_id=line_id,
                event_code=current_event_code,
                event_state="CLEARED",
                event_ts=event_ts,
                reading_id=reading_id,
                replay_cycle_id=replay_cycle_id,
                moisture_value=moisture_value,
                previous_event_code=current_event_code,
                next_event_code=new_event_code,
            )
        )

    if transitioned and new_event_code is not None:
        events.append(
            _insert_alarm_event(
                connection,
                line_id=line_id,
                event_code=new_event_code,
                event_state="ACTIVE",
                event_ts=event_ts,
                reading_id=reading_id,
                replay_cycle_id=replay_cycle_id,
                moisture_value=moisture_value,
                previous_event_code=current_event_code,
                next_event_code=new_event_code,
            )
        )

    connection.execute(
        UPDATE_ALARM_STATE_SQL,
        {
            "line_id": line_id,
            "rule_key": RULE_KEY,
            "replay_cycle_id": replay_cycle_id,
            "state": (
                "ACTIVE"
                if new_event_code is not None
                else "NORMAL"
            ),
            "severity": (
                "WARNING"
                if new_event_code is not None
                else None
            ),
            "active_event_code": new_event_code,
            "reading_id": reading_id,
            "event_ts": event_ts,
            "transitioned": transitioned,
        },
    )

    return events


def save_telemetry(payload: dict) -> dict | None:
    parameters = _validate_payload(payload)

    with engine.begin() as connection:

        result = connection.execute(
            INSERT_TELEMETRY_SQL,
            parameters,
        )

        reading = result.mappings().one_or_none()

        if reading is None:
            return None

        reading_id = reading["id"]
        received_at = reading["received_at"]

        events = []

        moisture_value = _get_moisture_value(payload)

        if moisture_value is not None:
            events = _process_moisture_alarm(
                connection,
                line_id=parameters["line_id"],
                event_ts=parameters["event_ts"],
                reading_id=reading_id,
                replay_cycle_id=parameters["replay_cycle_id"],
                moisture_value=moisture_value,
            )

        return {
            "id": reading_id,
            "received_at": received_at.isoformat(),
            "events": events,
        }

def get_hourly_summary(
    line_id: str,
    hours: int | None = None,
    cycle_id: str | None = None,
    hour_start: datetime | None = None,
) -> dict:
    result = get_hourly_summary_data(line_id, hours, cycle_id, hour_start)

    if result is None:
        return {
            "cycle_id": None,
            "window_start": None,
            "window_end": None,
            "hours": hours,
            "sample_count": 0,
            "data": [],
            "statistics": {},
        }

    cycle_id = result["cycle_id"]
    window_start = result["window_start"]
    window_end = result["window_end"]
    stats = result["statistics"]
    prefixes = {
        "oil_temperature": "burner_pv",
        "delta_t": "delta_t",
        "burner_load": "burner_load",
        "moisture": "moisture",
        "paddle_speed": "paddle_speed",
        "oil_level": "oil_level",
    }
    statistics = {
        key: {
            "average": _number(stats[f"{prefix}_avg"]),
            "minimum": _number(stats[f"{prefix}_min"]),
            "maximum": _number(stats[f"{prefix}_max"]),
        }
        for key, prefix in prefixes.items()
    }
    data = []
    for row in result["hourly_rows"]:
        item = {
            "hour_start": row["hour_start"].isoformat(),
            "sample_count": row["sample_count"],
            "burner_at_100_seconds": float(row["burner_at_100_seconds"]),
        }
        for prefix in prefixes.values():
            item[f"{prefix}_average"] = _number(row[f"{prefix}_avg"])
            item[f"{prefix}_minimum"] = _number(row[f"{prefix}_min"])
            item[f"{prefix}_maximum"] = _number(row[f"{prefix}_max"])
        data.append(item)

    return {
        "cycle_id": str(cycle_id),
        "window_start": window_start.isoformat(),
        "window_end": (window_end + timedelta(hours=1)).isoformat(),
        "hours": len(data),
        "sample_count": stats["sample_count"],
        "data": data,
        "statistics": statistics,
    }

def parse_timestamp(value, parameter_name):
    if not value:
        return None

    try:
        timestamp = datetime.fromisoformat(value)
    except (TypeError, ValueError):
        raise ValueError(
            f"{parameter_name} must be a valid ISO 8601 timestamp."
        ) from None

    if timestamp.tzinfo is not None:
        raise ValueError(
            f"{parameter_name} must be timezone-naive."
        )

    return timestamp

import logging

from server.extensions import socketio

logger = logging.getLogger(__name__)


def broadcast_telemetry(record, payload):
    line_id = payload["line_id"]

    event_data = {
        "id": record["id"],
        "message_id": payload["message_id"],
        "replay_session_id": payload["replay_session_id"],
        "replay_cycle_id": payload["replay_cycle_id"],
        "replay_cycle_number": payload["replay_cycle_number"],
        "line_id": line_id,
        "event_ts": payload["event_ts"],
        "received_at": record["received_at"],
        "payload": payload,
    }

    socketio.emit(
        "telemetry_update",
        event_data,
        to=f"line:{line_id}",
    )

    logger.debug(
        "Telemetry broadcast: record_id=%s line_id=%s",
        record["id"],
        line_id,
    )


def broadcast_operational_events(record):
    for event in record["events"]:
        socketio.emit(
            "operational_event",
            event,
            to=f"line:{event['line_id']}",
        )

        logger.info(
            "Operational event broadcast: "
            "event_id=%s code=%s state=%s",
            event["id"],
            event["event_code"],
            event["state"],
        )

from flask_socketio import join_room, leave_room

from server.extensions import socketio


ALLOWED_LINE_IDS = {"line30"}


@socketio.on("join_line")
def handle_join_line(data):
    if not isinstance(data, dict):
        return {
            "status": "error",
            "message": "A line_id is required.",
        }

    line_id = data.get("line_id")

    if not isinstance(line_id, str) or line_id not in ALLOWED_LINE_IDS:
        return {
            "status": "error",
            "message": "Unknown or unauthorized line.",
        }

    room = f"line:{line_id}"
    join_room(room)

    return {
        "status": "joined",
        "line_id": line_id,
        "room": room,
    }


@socketio.on("leave_line")
def handle_leave_line(data):
    if not isinstance(data, dict):
        return {
            "status": "error",
            "message": "A line_id is required.",
        }

    line_id = data.get("line_id")

    if not isinstance(line_id, str) or line_id not in ALLOWED_LINE_IDS:
        return {
            "status": "error",
            "message": "Unknown or unauthorized line.",
        }

    room = f"line:{line_id}"
    leave_room(room)

    return {
        "status": "left",
        "line_id": line_id,
        "room": room,
    }
